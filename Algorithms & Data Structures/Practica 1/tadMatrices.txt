El TAD Matriz es un tipo de dato 

Y tiene el sig. conjunto de operaciones

crearMatriz:  -> Matriz

/* Pre-condición: r ha sido creado. */

sumar: Matriz X Matriz -> Matriz
/* Pre-condición:  m1 y m2 han sido creados. */

restar: Matriz X Matriz -> Matriz
/* Pre-condición:  m1 y m2 han sido creados. */

multiplicar: Matriz X Matriz -> Matriz
/* Pre-condición:  m1 y m2 han sido creados. */ 

dividir:  Matriz X (Matriz -{0})-> Matriz
/* Pre-condición:  m1 y m2 han sido creados. y r2 <> 0*/

//Hacer una copia de un Matriz.
copia: Matriz -> Matriz

//Saber si dos Matrizes son iguales.
sonIguales: Matriz X Matriz -> boolean
